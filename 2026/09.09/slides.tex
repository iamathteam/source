\section{AMC-Style Practice}

\begin{frame}[t]{Question 1}
    What is the value of $\sqrt{\left(3-2\sqrt{3}\right)^2}+\sqrt{\left(3+2\sqrt{3}\right)^2}$?
\end{frame}

\begin{frame}[t]{Question 2}
    Brilliant Ben Shelton hits a serve at 225 kilometers per hour, while  Slow Sreyas Sabbani hits a serve at 20 kilometers per hour. What percent faster is Ben’s serve than Sreyas’s?
\end{frame}
 
\begin{frame}[t]{Question 3}
     Rohan, Sreyas, and Varun are prisoners each wearing one hat that is either yellow or pink. At one moment, each person chooses to either guess the color of their hat or stay silent. If at least one person guesses their hat color correctly, and no one guesses wrong, then they are released. Otherwise, they are sentenced to death. Find an optimal strategy so the three prisoners can hopefully survive.
\end{frame}

\begin{frame}[t]{Question 4}
    What is the units digit of \[ \underbrace{2027^{2027^{\cdot^{\cdot^{\cdot^{2027}}}}}}_{\text{2027 times}}\]
\end{frame}

\begin{frame}[t]{Question 5}
    Aidan and Hutchinson flip coins. They flip coins until they get one head. What is the probability that Aidan flips the coin less times than Hutchinson?
\end{frame}

\begin{frame}[t]{Question 6}
    Eminem starts rapping at 60 words per minute for two minutes. Then at the chorus he halves his pace for a minute and continues rapping now at 90 wpm for 3 minutes. What is his average wpm?
\end{frame}

\begin{frame}[t]{Question 7}
    What is the greatest number of consecutive integers whose sum is $45$?

    \vspace{3cm}
    
    Follow-up: What is the greatest number of consecutive non-negative integers whose sum is $45$?
\end{frame}

\begin{frame}[t]{Question 8}
    Positive real numbers $b \not= 1$ and $n$ satisfy the equations\[\sqrt{\log_b n} = \log_b \sqrt{n} \qquad \text{and} \qquad b \cdot \log_b n = \log_b (bn).\]The value of $n$ is $\frac{j}{k},$ where $j$ and $k$ are relatively prime positive integers. Find $j+k.$
\end{frame}
